data Int = Int;
data Bool = True | False;
data Maybe a = Just a | Nothing;
data Either a b = Left a | Right b;
data Pair a b = MkPair a b;
data List a = Cons a (List a) | Nil;
data Tree a = Leaf | Branch (Tree a) a (Tree a);

not : Bool -> Bool;
not b =
    case b of
        True => False
        | False => True;

filter : forall a. (a -> Bool) -> [a] -> [a];
filter f xs = 
    case xs of
        [] => []
        | y :: ys =>
         if f y then y :: filter f ys
         else filter f ys;

map : forall a b. (a -> b) -> [a] -> [b];
map f xs = 
    case xs of
        [] => []
        | y :: ys => (f y) :: map f ys;

intRange : Int -> Int -> Bool -> [Int];
intRange start end inclusive =
    if start > end then []
    else if start == end then
        if inclusive then [start] else []
    else
        start :: intRange (start + 1) end inclusive;

foldr : forall a b. (a -> b -> b) -> b -> [a] -> b;
foldr f z xs =
    case xs of
        [] => z
        | y :: ys => f y (foldr f z ys);

foldl : forall a b. (b -> a -> b) -> b -> [a] -> b;
foldl f acc xs =
    case xs of
        [] => acc
        | y :: ys => foldl f (f acc y) ys;

append : forall a. [a] -> [a] -> [a];
append xs ys =
    case xs of
        [] => ys
        | x :: rest => x :: append rest ys;

treeInsert : Int -> Tree Int -> Tree Int;
treeInsert x t =
    case t of
        Leaf => Branch Leaf x Leaf
        | Branch left y right =>
            if x <= y then
                Branch (treeInsert x left) y right
            else
                Branch left y (treeInsert x right);

treeFromList : [Int] -> Tree Int;
treeFromList xs =
    foldl (fun t x => treeInsert x t) Leaf xs;

treeToList : Tree Int -> [Int];
treeToList t =
    case t of
        Leaf => []
        | Branch left x right =>
            append (treeToList left) (x :: treeToList right);

listSample = intRange 1 500 True;

main = 
    let tree = treeFromList listSample in
        foldl (fun acc x => acc + x) 0 (treeToList tree);
